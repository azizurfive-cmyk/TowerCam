// ================================
// SEND TELEGRAM MESSAGE
// ================================

function sendMessage(chatId, text) {

  UrlFetchApp.fetch(TELEGRAM_URL + "/sendMessage", {

    method: "post",

    contentType: "application/json",

    payload: JSON.stringify({

      chat_id: chatId,

      text: text,

      parse_mode: "HTML",

      disable_web_page_preview: true

    })

  });
}



// ================================
// FIRESTORE GET TOWER DATA
// ================================

function getFirestoreData(documentId) {

  try {

    const url =
      "https://firestore.googleapis.com/v1/projects/" +
      PROJECT_ID +
      "/databases/(default)/documents/azizur/" +
      documentId;

    const response = UrlFetchApp.fetch(url, {
      muteHttpExceptions: true
    });

    if (response.getResponseCode() !== 200) {
      return null;
    }

    const data = JSON.parse(response.getContentText());

    if (data.fields && data.fields.pin) {
      return data.fields.pin.stringValue;
    }

    return null;

  } catch (e) {

    Logger.log("Tower Lookup Error: " + e.toString());

    return null;
  }
}



// ================================
// GET FULL USER DATA
// ================================

function getUserData(userId) {

  try {

    const url =
      "https://firestore.googleapis.com/v1/projects/" +
      PROJECT_ID +
      "/databases/(default)/documents/users/" +
      userId;

    const response = UrlFetchApp.fetch(url, {
      muteHttpExceptions: true
    });

    if (response.getResponseCode() !== 200) {
      return null;
    }

    const data = JSON.parse(response.getContentText());

    return {

      credit:
        data.fields.credit
          ? parseInt(data.fields.credit.integerValue)
          : 0,

      totalSearch:
        data.fields.totalSearch
          ? parseInt(data.fields.totalSearch.integerValue)
          : 0,

      successSearch:
        data.fields.successSearch
          ? parseInt(data.fields.successSearch.integerValue)
          : 0,

      failedSearch:
        data.fields.failedSearch
          ? parseInt(data.fields.failedSearch.integerValue)
          : 0

    };

  } catch (e) {

    Logger.log("Get User Data Error: " + e.toString());

    return null;
  }
}



// ================================
// UPDATE USER STATS
// ================================

function updateUserStats(userId, userData, isSuccess) {

  try {

    const url =
      "https://firestore.googleapis.com/v1/projects/" +
      PROJECT_ID +
      "/databases/(default)/documents/users/" +
      userId +
      "?updateMask.fieldPaths=credit" +
      "&updateMask.fieldPaths=totalSearch" +
      "&updateMask.fieldPaths=successSearch" +
      "&updateMask.fieldPaths=failedSearch" +
      "&updateMask.fieldPaths=lastActivity";



    const newCredit =
      userData.credit - 1;

    const newTotalSearch =
      userData.totalSearch + 1;

    const newSuccessSearch =
      isSuccess
        ? userData.successSearch + 1
        : userData.successSearch;

    const newFailedSearch =
      isSuccess
        ? userData.failedSearch
        : userData.failedSearch + 1;



    const updateData = {

      fields: {

        credit: {
          integerValue: String(newCredit)
        },

        totalSearch: {
          integerValue: String(newTotalSearch)
        },

        successSearch: {
          integerValue: String(newSuccessSearch)
        },

        failedSearch: {
          integerValue: String(newFailedSearch)
        },

        lastActivity: {
          stringValue: new Date().toISOString()
        }

      }
    };



    UrlFetchApp.fetch(url, {

      method: "patch",

      contentType: "application/json",

      payload: JSON.stringify(updateData)

    });



    return newCredit;

  } catch (e) {

    Logger.log("Update Stats Error: " + e.toString());

    return userData.credit;
  }
}

// ================================
// LOG NOT FOUND SITES TO FIRESTORE
// ================================
function logNotFoundToFirestore(userId, userName, siteCode) {
  try {
    const logId = "log_" + new Date().getTime() + "_" + userId;
    const url =
      "https://firestore.googleapis.com/v1/projects/" +
      PROJECT_ID +
      "/databases/(default)/documents/not_found_logs/" +
      logId;

    const logData = {
      fields: {
        userId: { stringValue: String(userId) },
        userName: { stringValue: userName || "Unknown" },
        siteCode: { stringValue: siteCode },
        timestamp: { stringValue: new Date().toISOString() }
      }
    };

    UrlFetchApp.fetch(url, {
      method: "patch",
      contentType: "application/json",
      payload: JSON.stringify(logData)
    });
  } catch (e) {
    Logger.log("Firestore Logging Error: " + e.toString());
  }
}

// ================================
// MAIN WEBHOOK
// ================================

function doPost(e) {

  try {

    const data = JSON.parse(e.postData.contents);



    if (!data.message || data.message.from.is_bot) {
      return;
    }



    const chatId = data.message.chat.id;

    const userId = data.message.from.id;

    const userName =
      data.message.from.first_name || "Unknown";



    // ================================
    // INPUT CLEAN
    // ================================

    // ইউজারের পাঠানো মূল টেক্সটটি এখানে রাখা হলো (স্পেসসহ)
    let rawInput = data.message.text ? data.message.text.trim() : "";
    let userInput = rawInput.toUpperCase();



    // ================================
    // /START
    // ================================

    if (userInput === "/START") {

      const startMsg =

        "📡 <b>Tower Location Bot</b>\n\n" +

        "একটি Site Code পাঠান\n\n" +

        "✅ উদাহরণ:\n" +

        "<code>DHBDD01</code>";



      sendMessage(chatId, startMsg);

      return;
    }



    // ================================
    // EMPTY OR OTHER COMMAND
    // ================================

    if (
      userInput === "" ||
      userInput.startsWith("/")
    ) {

      const helpMsg =

        "📡 একটি সঠিক Site Code পাঠান\n\n" +

        "✅ উদাহরণ:\n" +

        "<code>DHBDD01</code>";



      sendMessage(chatId, helpMsg);

      return;
    }



    // ================================
    // SPACE / MULTI-WORD VALIDATION (FIXED)
    // ================================
    // টেক্সটের মাঝে কোনো স্পেস থাকলে ডাটাবেজে যাওয়ার আগেই আটকে দেবে
    if (/\s/g.test(rawInput)) {
      const spaceWarnMsg = 
        "⚠️ <b>ভুল সাইট কোড!</b>\n\n" +
        "আমাদের সাইট কোড এক শব্দের হয়, মাঝে কোনো স্পেস (Space) বা একাধিক শব্দ ব্যবহার করা যাবে না।\n\n" +
        "✅ সঠিক উদাহরণ:\n" +
        "<code>DHBDD01</code>";

      sendMessage(chatId, spaceWarnMsg);
      return; 
    }



    // ================================
    // ONLY ENGLISH VALIDATION
    // ================================

    if (/[^A-Za-z0-9_]/.test(userInput)) {

      const warnMsg =

        "⚠️ <b>শুধু English Site Code ব্যবহার করুন</b>\n\n" +

        "✅ উদাহরণ:\n" +

        "<code>DHBDD01</code>";



      sendMessage(chatId, warnMsg);

      return;
    }



    // ================================
    // LENGTH VALIDATION
    // ================================

    const len = userInput.length;



    // too short
    if (len < 4) {

      sendMessage(

        chatId,

        "⚠️ ৪ অক্ষরের কম কোনো Site Code হয় না।"

      );

      return;
    }



    // too long
    if (len > 15) {

      sendMessage(

        chatId,

        "🚫 Invalid Site Code"

      );

      return;
    }



    // ================================
    // FIRESTORE START
    // ================================



    // register user
    registerUserWithCredit(userId, userName);



    // get full user data
    const userData =
      getUserData(userId);



    // firestore fail
    if (userData === null) {

      sendMessage(

        chatId,

        "⚠️ Server Busy.\nপরে আবার চেষ্টা করুন।"

      );

      return;
    }



    // no credit
    if (userData.credit <= 0) {

      const noCreditMsg =

        "📡 <b>টাওয়ার লোকেশন ক্রেডিট শেষ!</b>\n\n" +

        "ভাই, আপনার অ্যাকাউন্টের <b>ফ্রি ক্রেডিট শেষ</b> হয়ে গেছে।\n\n" +

        "নতুন লোকেশন সার্চ করতে এখনই নতুন ক্রেডিট নিন।\n\n" +

        "🔄 <b>যোগাযোগ:</b>\n" +

        "<code>01886196651</code>\n\n" +

        "👉 <a href='https://t.me/Tower_Location'>🎯 ক্রেডিট নিতে ক্লিক করুন</a>\n\n" +

        "🆔 <b>User ID:</b> " +
        userId +
        "\n\n" +

        "<i>ধন্যবাদ ❤️</i>";



      sendMessage(chatId, noCreditMsg);

      return;
    }



    // ================================
    // TOWER LOOKUP
    // ================================

    const documentId = userInput;

    const pinData =
      getFirestoreData(documentId);



    // ================================
    // FOUND
    // ================================

    if (pinData) {

      // update stats
      const remainingCredit =
        updateUserStats(
          userId,
          userData,
          true
        );



      const coords =
        pinData.replace(/\s+/g, "").split(",");



      const lat = coords[0];

      const lon = coords[1];



      const mapLink =
        "https://www.google.com/maps/search/?api=1&query=" +
        lat +
        "," +
        lon;



      const replyText =

        "📡 <b>Site Code:</b> " +
        documentId +
        "\n\n" +

        "📍 <b>Lat, Lon: </b> " +

        "<code>" +
        lat +
        ", " +
        lon +
        "</code>\n\n" +

        "🌐 <b>Google Map:</b>\n" +
        mapLink +
        "\n\n" +

        "💳 ১ ক্রেডিট কাটা হয়েছে\n" +

        "💰 অবশিষ্ট ক্রেডিট: <b>" +
        remainingCredit +
        "</b>\n\n" +

        "👉 <a href='https://t.me/Tower_Location_bot'>🎯 Tower Location</a>";



      sendMessage(chatId, replyText);

      return;
    }



    // ================================
    // NOT FOUND
    // ================================

    // update stats
    const remainingCredit =
      updateUserStats(
        userId,
        userData,
        false
      );


      // ফায়ারস্টোরে লগ সেভ করার ফাংশনটি এখানে কল করা হলো
      logNotFoundToFirestore(userId, userName, documentId);


    const notFoundMsg =

      "❌ <b>আইডি পাওয়া যায়নি!</b>\n\n" +

      "🔍 দুঃখিত ভাই,\n" +

      "এই Site Code টি আমাদের ডাটাবেজে নেই:\n\n" +

      "<code>" +
      documentId +
      "</code>\n\n" +


      "🛠️ <b>আপনি কীভাবে সাহায্য করতে পারেন?</b>\n" +
      "Your information can make a big difference! 🙏✨\n\n" +
      "📢 <b>তথ্য আপডেট করার মাধ্যমসমূহ:</b>\n" +
      "🤖 <b>আমাদের বটের মাধ্যমে:</b> সরাসরি নিচের লিংকে গিয়ে নিজে নিজেই আপডেট করে দিন:\n" +
      "👉 @new_tower_location_update_bot 🔄\n\n" +
      "💬 <b>আমাদের গ্রুপে পোস্ট করুন:</b> অথবা আমাদের গ্রুপে টাওয়ারের তথ্য লিখে পোস্ট দিন, আমরা পরবর্তীতে তা কালেক্ট করে ডাটাবেজে যুক্ত করে নেব : https://t.me/new_tower_location_update_group \n" +
      "👉 @Tower_Location 👥\n\n" +
      "🤝 <i>সবার সম্মিলিত প্রচেষ্টায় গড়ে উঠুক একটি নির্ভুল নেটওয়ার্ক ডাটাবেজ। ধন্যবাদ! ❤️</i>";



    sendMessage(chatId, notFoundMsg);



  } catch (e) {

    Logger.log("Main Error: " + e.toString());
  }
}
